```typescript
/**
 * Narrow seam called synchronously inside per-episode db.transaction (D-48).
 * No async/await — better-sqlite3 is synchronous; an await between the graph
 * mutation and its emit would break the atomicity guarantee.
 */
export interface ConsolidationSink {
  emit(event: ConsolidationEventInput): void;
}

export class SQLiteConsolidationSink implements ConsolidationSink {
  private readonly eventStore: EventStore;
  private readonly clock: Clock;
  constructor(eventStore: EventStore, clock: Clock) { /* ... */ }
  emit(event: ConsolidationEventInput): void {
    this.eventStore.append({
      id: newId(), ts: this.clock.nowMs(), schema_version: SCHEMA_VERSION,
      event_type: event.event_type, node_id: event.node_id ?? null, /* ... */
    });
  }
}

export class NoopConsolidationSink implements ConsolidationSink {
  // eslint-disable-next-line @typescript-eslint/no-unused-vars
  emit(_event: ConsolidationEventInput): void {
    // Intentional no-op — used as the default when no sink is injected.
  }
}

export class MockConsolidationSink implements ConsolidationSink {
  readonly events: ConsolidationEventInput[] = [];
  emit(event: ConsolidationEventInput): void { this.events.push(event); }
  reset(): void { this.events.length = 0; }
}
```

<threat_model>
## Trust Boundaries

| Boundary | Description |
|----------|-------------|
| attacker-controlled email content → `evidence_quote` | Untrusted free text crosses into a persisted, later-served field |
| consolidator transaction → `action_proposal` insert | An engine write derived from model-classified input |

## STRIDE Threat Register

| Threat ID | Category | Component | Disposition | Mitigation Plan |
|-----------|----------|-----------|-------------|-----------------|
| T-66-06 | Elevation of Privilege | `evidence_quote` handling in `SQLiteActionProposalSink.emit` | mitigate | The quote is passed through as an opaque string — the module has no formatter, no summariser, and no `ModelProvider` parameter of any kind, so a hidden instruction inside it has no interpreter on this path. Byte-equality round-trip test with an injection-shaped payload is the proof |
| T-66-04 | Repudiation | `proposalId` | mitigate | Content hash over exactly five keyed fields with a locked `JSON.stringify` array encoding; timestamps and the randomly-minted `belief_node_id` are excluded, proven by dedicated sensitivity tests |
| T-66-10 | Tampering | id collision / preimage | accept | `sha256` over engine-internal identifiers; an attacker controls at most `change_to` (a closed `IntentStatus` token) and `evidence_episode` (an engine-minted id) — no attacker-steerable preimage surface. Documented rather than mitigated |
| T-66-11 | Denial of Service | unbounded `evidence_quote` | accept | Episode content is already capped at `maxContentBytes` (8KB) by `EpisodicStore.capContent` upstream, so the field is bounded by construction. No second cap is added, because a truncating cap would falsify EMIT-06's "verbatim" claim for the exact long-payload case Pitfall 2 cares about |
| T-66-12 | Tampering | default-on emission | mitigate | `actionProposalSinkEnabled` defaults `false` AND the Consolidator constructor default is `NoopActionProposalSink` — two independent barriers, so a wiring mistake in one place cannot silently start writing |
| T-66-SC | Tampering | package installs | accept | No package-manager install in this plan; `sha256` is the shipped `node:crypto` wrapper. Nothing for the Package Legitimacy Gate to audit |
</threat_model>
