```typescript
// clients/proposal-reference/config.ts
export interface AdapterConfig { serveUrl: string; serveToken: string; localStorePath: string; enabled: boolean; }
export function loadAdapterConfig(): AdapterConfig;

// clients/proposal-reference/proposal-client.ts
export const PROPOSAL_SCHEMA_VERSION: number; // 17
export interface ActionProposalRecord { /* 16 fields, snake_case, wire shape */ }
export interface ProposalClient {
  listProposals(): Promise<ActionProposalRecord[]>;
  approve(id: string): Promise<void>;   // throws ProposalHttpError { status } on non-2xx
  reject(id: string): Promise<void>;    // throws ProposalHttpError { status } on non-2xx
}
export function createProposalClient(serveUrl: string, serveToken: string): ProposalClient;

// clients/proposal-reference/local-store.ts
export type LocalStatus = 'pending' | 'applied' | 'refused' | 'skipped';
export interface LocalRow {
  localId: string; proposalId: string; entityDescriptor: string;
  changeField: string; changeTo: string;
  localStatus: LocalStatus; refusalReason: string | null; updatedAtMs: number;
}
export function newLocalId(): string;
export function listLocalRows(storePath: string): LocalRow[];
export function findByProposalId(proposalId: string, storePath: string): LocalRow | undefined;
export function putLocalRow(row: LocalRow, storePath: string): void;
```

Server status contract the loop branches on (numeric status only — never the `detail` string):

| Status | `error` | Loop behaviour |
|---|---|---|
| 200 | — | local row → `applied` |
| 400 | `bad_request` | terminal: local row → `refused`, `refusalReason: 'bad_request'` |
| 401 | `unauthorized` | abort the whole sync (auth is broken, not this item) |
| 404 | `not_found` | terminal: local row → `refused`, `refusalReason: 'not_found'` |
| 409 | `conflict` | terminal: local row → `refused`, `refusalReason: 'conflict'` — never retried |
| 503 | `service_unavailable` | NOT terminal: leave the local row `pending`, retry on a later sync |

<threat_model>
## Trust Boundaries

| Boundary | Description |
|----------|-------------|
| serve response body → loop control flow | `evidence_quote` / `change_from` are attacker-influenced free text arriving as data the loop must not act on |
| serve HTTP status → retry policy | Misclassifying a terminal refusal as retryable turns one bad proposal into an unbounded POST loop |
| local store file → "already applied?" decision | The idempotency verdict is read from disk state that a local attacker could edit |
| adapter log file | The only place the adapter writes free text; a naive log line could leak the Bearer token or echo injected prose |

## STRIDE Threat Register

| Threat ID | Category | Component | Disposition | Mitigation Plan |
|-----------|----------|-----------|-------------|-----------------|
| T-67-08 | Tampering (prompt injection) | `decideOutcome` / `syncProposals` | mitigate | `decideOutcome` reads only `confidence` and `kind`; `evidence_quote` and `change_from` are never read, never persisted, never printed by `list`, and never interpolated into a log line. Enforced by the comment-stripped grep gate in Task 1 and the injected-string test in Task 2 |
| T-67-09 | Tampering / Repudiation | refusal handling | mitigate | 400/404/409 write `localStatus: 'refused'` and are never re-POSTed; only 503 is treated as retryable. Proven by the "no second POST" assertion on the stub's recorded call list (Task 2) |
| T-67-10 | Tampering | replay / double-apply | mitigate | `findByProposalId` gate before every POST plus write-pending-before-POST ordering, so a crash mid-flight resumes rather than double-applies (D-02, EMIT-04 consumer half) |
| T-67-11 | Denial of service | batch loop | mitigate | Per-item `try/catch` (one failure never aborts the batch); no retry loop inside a single sync — a 503 defers to the next invocation rather than spinning; response size bounded server-side at `PROPOSAL_LIST_LIMIT = 100` |
| T-67-12 | Information disclosure | adapter log / `list` output | mitigate | The token is never logged; `list` prints only `id`, `entity_descriptor`, `change_field`, `change_to`, `confidence`, `expires_at`; the fatal-error handler appends the error, not the config |
| T-67-13 | Elevation of privilege | 401 handling | mitigate | A 401 aborts the entire sync rather than being swallowed per-item — a broken/rotated token must be loud, not silently skipped (identical-401 discipline, docs/reference-client.md:130) |
| T-67-SC | Tampering | npm installs | accept | Zero package-manager installs in this plan (`node:*` built-ins and existing devDeps only); `git diff package-lock.json` must be empty, so no legitimacy checkpoint applies |
</threat_model>
